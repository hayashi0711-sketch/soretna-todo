## 🔧 App.jsx への統合方法

### Step 1: Hooksのインポート

```javascript
import { useAuth } from '@/hooks/useAuth';
import { useGroups } from '@/hooks/useGroups';
import { useGroupTodos } from '@/hooks/useGroupTodos';
import { useNotifications } from '@/hooks/useNotifications';
import { useGroupMembers } from '@/hooks/useGroupMembers';
import { useActivityFeed } from '@/hooks/useActivityFeed';
```

### Step 2: コンポーネントのインポート

```javascript
import { GroupSelector } from '@/components/Groups/GroupSelector';
import { GroupModal } from '@/components/Groups/GroupModal';
import { SharedTaskForm } from '@/components/SharedTasks/SharedTaskForm';
import { SharedTaskList } from '@/components/SharedTasks/SharedTaskList';
import { NotificationPanel } from '@/components/Notifications/NotificationPanel';
```

### Step 3: App.jsx内でのHook初期化

```javascript
export default function App() {
  // 認証
  const { user, userProfile, loading: authLoading, signUp, signIn, signOut } = useAuth();
  
  // グループ
  const { groups, loading: groupsLoading, createGroup, updateGroup, deleteGroup } = useGroups(user?.uid);
  const { addMemberToGroup, removeMemberFromGroup, updateMemberRoleInGroup } = useGroupMembers();
  
  // 選択中のグループ
  const [selectedGroupId, setSelectedGroupId] = useState(null);
  
  // グループ内タスク
  const { todos: groupTodos, loading: todosLoading, addTodo, updateTodo, deleteTodo, toggleTodo } = useGroupTodos(selectedGroupId);
  
  // 通知
  const { notifications, unreadCount, loading: notificationsLoading, addNotification, markAsRead, deleteNotification, markAllAsRead } = useNotifications(user?.uid);
  
  // アクティビティフィード
  const { activities, loading: activitiesLoading, addActivity } = useActivityFeed(selectedGroupId);

  // グループ作成ハンドラ
  const handleCreateGroup = async (formData) => {
    try {
      const groupId = await createGroup(formData.name, formData.description);
      setSelectedGroupId(groupId);
      // グループ作成通知
      await addNotification({
        userId: user.uid,
        type: 'group_created',
        groupId,
        message: `グループ「${formData.name}」を作成しました`,
        triggeredBy: user.uid,
        triggeredByName: userProfile?.displayName,
        read: false,
        createdAt: new Date(),
      });
    } catch (err) {
      console.error('Error creating group:', err);
    }
  };

  // タスク追加ハンドラ
  const handleAddGroupTask = async (taskData) => {
    try {
      const taskId = await addTodo(selectedGroupId, taskData.text, taskData.priority, taskData.tags);
      
      // タスク作成通知をグループメンバーに送信
      await addNotification({
        userId: user.uid,
        type: 'todo_created',
        groupId: selectedGroupId,
        message: `「${taskData.text}」がグループに追加されました`,
        triggeredBy: user.uid,
        triggeredByName: userProfile?.displayName,
        relatedTodoId: taskId,
        read: false,
        createdAt: new Date(),
      });
      
      // アクティビティに記録
      await addActivity({
        userId: user.uid,
        userName: userProfile?.displayName,
        type: 'todo_created',
        description: `${taskData.text} を作成しました`,
        relatedTodoId: taskId,
        createdAt: new Date(),
      });
    } catch (err) {
      console.error('Error adding group task:', err);
    }
  };

  // タスク完了トグル
  const handleToggleGroupTask = async (taskId) => {
    try {
      await toggleTodo(selectedGroupId, taskId, user.uid);
      
      // 完了通知
      const todo = groupTodos.find(t => t.id === taskId);
      if (todo) {
        await addActivity({
          userId: user.uid,
          userName: userProfile?.displayName,
          type: todo.done ? 'todo_completed' : 'todo_updated',
          description: `「${todo.text}」を${todo.done ? '完了' : '再開'}しました`,
          relatedTodoId: taskId,
          createdAt: new Date(),
        });
      }
    } catch (err) {
      console.error('Error toggling group task:', err);
    }
  };

  return (
    <div style={{ ... }}>
      {/* ログイン画面 */}
      {!user ? (
        <LoginScreen onSignUp={signUp} onSignIn={signIn} />
      ) : (
        <>
          {/* ヘッダー */}
          <header style={{ ... }}>
            <h1>それな！Todo 📋</h1>
            <div style={{ display: 'flex', gap: '12px', alignItems: 'center' }}>
              <NotificationPanel
                notifications={notifications}
                unreadCount={unreadCount}
                onMarkAsRead={markAsRead}
                onDeleteNotification={deleteNotification}
                onMarkAllAsRead={markAllAsRead}
              />
              <span>{userProfile?.displayName}</span>
              <button onClick={signOut}>ログアウト</button>
            </div>
          </header>

          {/* メインコンテンツ */}
          <main style={{ ... }}>
            {/* グループセレクタ */}
            <GroupSelector
              groups={groups}
              selectedGroupId={selectedGroupId}
              onSelectGroup={setSelectedGroupId}
              onCreateGroup={() => {/* モーダル表示 */}}
              onOpenSettings={() => {/* 設定画面 */}}
            />

            {selectedGroupId ? (
              <>
                {/* グループ内タスク */}
                <div>
                  <h2>グループタスク</h2>
                  <SharedTaskForm
                    selectedGroupId={selectedGroupId}
                    onAddTask={handleAddGroupTask}
                  />
                  <SharedTaskList
                    todos={groupTodos}
                    loading={todosLoading}
                    onToggleTodo={handleToggleGroupTask}
                    onDeleteTodo={(taskId) => deleteTodo(selectedGroupId, taskId)}
                    currentUserId={user.uid}
                  />
                </div>
              </>
            ) : (
              <div style={{ ... }}>
                グループを選択してください
              </div>
            )}

            {/* 既存の個人タスク */}
            <div>
              <h2>個人タスク</h2>
              {/* 既存のタスク表示コンポーネント */}
            </div>
          </main>
        </>
      )}

      {/* グループ作成モーダル */}
      <GroupModal
        isOpen={isGroupModalOpen}
        onClose={() => setIsGroupModalOpen(false)}
        onSubmit={handleCreateGroup}
      />
    </div>
  );
}
```

## 💡 トラブルシューティング

### Issue 1: "firebaseのモジュールが見つからない"
```
解決: npm install firebase
```

### Issue 2: "Firestore へのアクセスが拒否される"
```
確認事項:
1. firestore.rules が正しくデプロイされているか
2. Firebase Console で認証が有効か
3. セキュリティルールで members フィールドを確認
```

### Issue 3: リアルタイム更新が反映されない
```
確認事項:
1. useGroupTodos の onSnapshot が購読している
2. selectedGroupId が正しく設定されている
3. ブラウザのネットワークが接続している
4. Firebase Console で インデックス作成が完了しているか
```

### Issue 4: 通知が表示されない
```
確認事項:
1. Notification インターフェースで userId と groupId を指定
2. NotificationPanel が notifications prop を受け取っている
3. useNotifications が userId で購読している
```
